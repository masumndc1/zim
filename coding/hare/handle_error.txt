-- What ! Does
-- When a function returns a tagged union that includes an error
	(like (void | error) or (int | fs::error)), appending ! tells the compiler:
	- If it's a success: Unwrap the underlying value and continue.
    - If it's an error: Immediately abort/crash the program with that error.
-- never use ! in type definitions, like

   $ haredoc errors::noaccess
   type noaccess = !void;

-- What ? Does
-- When you write const file = os::create(path, 0o644, oflags)?;, it means:
   	- If it succeeds: Unwrap the io::file and assign it to file.
	- If it fails (returns an error): Immediately return that error
	- out of the current function to whoever called without crashing the programme.

-- Think of it as saying: "I am not crashing the program here;
-- I'm passing the problem up to my caller to deal with."

-- when to use free

-- in haredoc if you see nomem in return type, then you need to use free to
	return the heap after using it.

 haredoc strings::concat
	fn concat (strs: str...) (str | nomem);

use fmt;
use strings;

export fn main() void = {
    let first = "Hello, ";
    let second = "Hare!";

    // strings::concat allocates a brand new string on the heap.
    // Because it allocates, its return type is: (str | nomem)
    let combined = match (strings::concat(first, second)) {
        case let s: str => yield s;
		case nomem => {
            fmt::errorfln("Error: Out of memory during concatenation!")!;
            return;
        };
    };

    // Since 'combined' is a fresh heap allocation that we now own,
    // we must defer free() to release the memory when main() exits.
    defer free(combined);

    fmt::printfln("Result: {}", combined)!;
};

-- if you dont free the memory, it can memory leak and oom kill the programme.
