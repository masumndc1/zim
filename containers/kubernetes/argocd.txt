

-- argo cd

-- assume you have kind cluster installed locally

-- now install argo cd

kubectl create namespace argocd
kubectl apply -n argocd --server-side --force-conflicts \
  -f https://raw.githubusercontent.com/argoproj/argo-cd/stable/manifests/install.yaml

-- list the argo cd pod
kubectl get pods -n argocd

-- Access the Argo CD Web UI via Port Forwarding
kubectl port-forward svc/argocd-server -n argocd 8080:443

-- login to following

https://localhost:8080

-- retrive the initial password where the username is admin

kubectl get secret argocd-initial-admin-secret -n argocd \
  -o jsonpath="{.data.password}" | base64 --decode \
  echo

-- also argocd cli can be installed by brew
brew install argocd
