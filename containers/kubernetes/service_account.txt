

-- service account
-- when another tools or service needs to authenticate itself rather than an user
-- service account can be used

❯ kubectl get serviceaccount
NAME      SECRETS   AGE
default   0         18d

❯
❯ kubectl create serviceaccount build-bot
serviceaccount/build-bot created

❯ kubectl get serviceaccount
NAME        SECRETS   AGE
build-bot   0         1s
default     0         18d
❯

❯ kubectl describe serviceaccount build-bot
Name:                build-bot
Namespace:           default
Labels:              <none>
Annotations:         <none>
Image pull secrets:  <none>
Events:              <none>
❯

-- this can be also done by from a yaml file

❯ cat service_pod.yaml
apiVersion: v1
kind: Pod
metadata:
  name: build-observer
spec:
  serviceAccountName: build-bot
  restartPolicy: Never
  containers:
  - name: observer
    image: alpine

-- role creation

❯ kubectl create role read-only --verb=list,get,watch --resource=pods,deployments,services
role.rbac.authorization.k8s.io/read-only created

❯ kubectl get role
NAME        CREATED AT
read-only   2026-11-10T17:29:13Z

❯ kubectl describe role read-only
Name:         read-only
Labels:       <none>
Annotations:  <none>
PolicyRule:
  Resources         Non-Resource URLs  Resource Names  Verbs
  ---------         -----------------  --------------  -----
  pods              []                 []              [list get watch]
  services          []                 []              [list get watch]
  deployments.apps  []                 []              [list get watch]

-- after we create the role we need to do the role bindings

> kubectl created rolebinding read-only-binding --role=read-only --user=user1
