-- image creation can be done also by following

wget http://download.cirros-cloud.net/0.4.0/cirros-0.4.0-x86_64-disk.img

glance image-create --name "cirros" \
  --file cirros-0.4.0-x86_64-disk.img \
  --disk-format qcow2 --container-format bare \
  --visibility=public

-- creating a image from iso file by following

glance image-create --name ISO_IMAGE --file IMAGE.iso \
  --disk-format iso --container-format bare

-- update the image metadata

curl -i -X PATCH \
  -H "X-Auth-Token: $TOKEN" \
  -H "Content-Type: application/openstack-images-v2.1-json-patch" \
  -d '[
    {"op": "replace", "path": "/name", "value": "debian-12-updated"},
    {"op": "add", "path": "/hw_firmware_type", "value": "uefi"}
  ]' \
  http://<Glance_Server_Location>:9292/v2/images/<ID>

-- same to delete an image

curl -i -X DELETE \
  -H "X-Auth-Token: $TOKEN" \
  http://<Glance_Server_Location>:9292/v2/images/<ID>

-- to sort by name and status

curl -s -H "X-Auth-Token: $token" \
  "https://glance-server:9293/v2/images?sort=name,status" | jq '.'

-- how to update an image properties

glance image-update \
    --property hw_disk_bus=scsi \
    --property hw_cdrom_bus=ide \
    --property hw_vif_model=e1000 \
    <Image-ID>

-- from https://docs.openstack.org/glance/latest/admin/manage-images.html

"Currently the libvirt virtualization tool determines the disk, CD-ROM,
and VIF device models based on the configured hypervisor type
(libvirt_type in /etc/nova/nova.conf file). For the sake of optimal performance,
libvirt defaults to using virtio for both disk and VIF (NIC) models.
The disadvantage of this approach is that it is not possible to run
operating systems that lack virtio drivers, for example,
BSD, Solaris, and older versions of Linux and Windows."
