-- when you run firefox or other programme.
-- higher cpu load:
-- try to do following

-- Enable Power Daemon (powerd):
-- Dynamically scales your CPU clock speed based on load.
-- Add this line to /etc/rc.conf:

powerd_enable="YES"
powerd_flags="-a adaptive -b adaptive"

-- Enable Deeper CPU Idle States (C-States):
-- Keeps idle CPU cores cool.
-- Add this to /etc/sysctl.conf:

hw.acpi.cpu.cx_lowest=Cmax
dev.cpu.0.cx_lowest=Cmax
dev.cpu.1.cx_lowest=Cmax
