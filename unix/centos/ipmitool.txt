-- another way of getting mac address

/opt/dell/srvadmin/sbin/racadm -r node.domain.com -u root -p calvin get NIC.VndrConfigPage.1.MacAddr --nocertwarn | grep MacAddr

-- alternatively, you can put the command in a bash file like

#!/bin/bash

for a in {120..135}; do
        host node$a.domain.com | cut -f1,5 -d ' '
        /opt/dell/srvadmin/sbin/racadm -r node$a-ipmi.domain.com \
		 -u user -p passwrd get NIC.VndrConfigPage.1.MacAddr \
		 --nocertwarn | grep MacAddr | cut -f2 -d '='
done
