-- solution:

sudo modprobe -r jme
sudo modprobe jme pcie_aspm=0 force_sel_100m=0

-- make it permanent

echo "options jme pcie_aspm=0 force_sel_100m=0" | sudo tee /etc/modprobe.d/jme.conf

-- another way. this worked for me.

sudo vi /etc/systemd/system/jme-phy-fix.service

[Unit]
Description=Fix JMicron JMC250 PHY Link Settings
After=sys-subsystem-net-devices-enp7s0f5.device
Wants=sys-subsystem-net-devices-enp7s0f5.device

[Service]
Type=oneshot
ExecStart=/usr/bin/ip link set enp7s0f5 down
ExecStart=/usr/sbin/ethtool -s enp7s0f5 speed 100 duplex full autoneg on
ExecStart=/usr/bin/ip link set enp7s0f5 up
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target

sudo systemctl daemon-reload
sudo systemctl enable jme-phy-fix.service

-- following also worked for me

# 1. Bring interface down
sudo ip link set enp7s0f5 down

# 2. Force 100Mbps Full Duplex with autonegotiation enabled
sudo ethtool -s enp7s0f5 speed 100 duplex full autoneg on

# 3. Bring interface up
sudo ip link set enp7s0f5 up
